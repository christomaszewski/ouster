cmake_minimum_required(VERSION 3.18)
project(ouster_gpu_cloud_prototype LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_EXPORTER "Build SDK/ROS offline packet exporter" OFF)
option(BUILD_CUDA_BENCHMARK "Build CUDA benchmark" OFF)
if(BUILD_EXPORTER)
  find_package(ament_cmake REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(pcl_conversions REQUIRED)
  find_package(pcl_msgs REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(message_filters REQUIRED)
  find_path(PCL_CONVERSIONS_INCLUDE_DIR pcl_conversions/pcl_conversions.h
    HINTS "${pcl_conversions_DIR}/../../../include" PATH_SUFFIXES pcl_conversions REQUIRED)
  find_package(PCL REQUIRED COMPONENTS common)
  find_package(Eigen3 REQUIRED)
  if(NOT ROS_SOURCE)
    message(FATAL_ERROR "ROS_SOURCE must identify the pinned ouster-ros package directory")
  endif()
  if(SDK_BUILD)
    # The retained ROS test image has the exact production archive, but its
    # build-tree OusterSDKConfig is not an installed/exported SDK package.
    find_package(Threads REQUIRED)
    find_package(OpenSSL REQUIRED)
    find_package(CURL REQUIRED)
    find_library(ZIP_LIBRARY NAMES zip REQUIRED)
    add_library(OusterSDK::ouster_client STATIC IMPORTED)
    set_target_properties(OusterSDK::ouster_client PROPERTIES
      IMPORTED_LOCATION "${SDK_BUILD}/ouster_client/libouster_client.a"
      INTERFACE_INCLUDE_DIRECTORIES "${ROS_SOURCE}/ouster-sdk/ouster_client/include;${ROS_SOURCE}/ouster-sdk/ouster_client/include/optional-lite"
      INTERFACE_LINK_LIBRARIES "${SDK_BUILD}/ouster_client/nmea/libnmea.a;${ZIP_LIBRARY};OpenSSL::SSL;OpenSSL::Crypto;CURL::libcurl;Threads::Threads")
  else()
    find_package(OusterSDK REQUIRED)
  endif()
  foreach(tool IN ITEMS export_scan cpu_production_benchmark)
    add_executable(${tool} ${tool}.cpp)
    target_include_directories(${tool} PRIVATE ${ROS_SOURCE}/src ${ROS_SOURCE}/include
      ${PCL_INCLUDE_DIRS} ${PCL_CONVERSIONS_INCLUDE_DIR})
    target_link_libraries(${tool} OusterSDK::ouster_client Eigen3::Eigen pcl_common
      ${sensor_msgs_TARGETS} ${pcl_msgs_TARGETS} rclcpp::rclcpp message_filters::message_filters)
    target_compile_options(${tool} PRIVATE -O3 -DNDEBUG)
  endforeach()
endif()
if(BUILD_CUDA_BENCHMARK)
  enable_language(CUDA)
  add_executable(cloud_benchmark cloud_benchmark.cu)
  set_target_properties(cloud_benchmark PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 87 CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(cloud_benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--ftz=false;--prec-div=true;--prec-sqrt=true>)
endif()
